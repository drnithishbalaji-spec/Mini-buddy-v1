#!/usr/bin/env python3 # ============================================================ # MINI AI BOT 🤖 # A tiny chatbot that uses Gemini or GPT as its brain. # # Commands (type these when the bot is running): # ask chat with the AI # website build a web page, saves website.html # picture draw a picture, saves picture.png # video write a video plan, saves video-plan.txt # use gemini switch brain to Gemini # use gpt switch brain to GPT # help show this list again # quit say goodbye # # You need 3 things (see README.txt for baby steps): # 1. Python on your computer (this does NOT run on a phone) # 2. pip install requests # 3. Your own API key pasted below (the free Gemini key is enough) # ============================================================ import base64 import datetime import sys try: import requests except ImportError: print("This bot needs the 'requests' package.") print("Install it with: pip install requests") print("(On Windows try: py -m pip install requests)") sys.exit(1) # ============================================================ # YOUR SETTINGS — the only part you ever need to edit # ============================================================ # Your Gemini key (FREE). Get one here: https://aistudio.google.com/apikey # Paste it between the quotes, replacing PASTE_YOUR_GEMINI_KEY_HERE. # Keep it secret — never send it to anyone, not even in chat. GEMINI_API_KEY = "PASTE_YOUR_GEMINI_KEY_HERE" # Your OpenAI key (needs a PAID OpenAI account). # Get one here: https://platform.openai.com/api-keys OPENAI_API_KEY = "PASTE_YOUR_OPENAI_KEY_HERE" # Which brain to start with: "gemini" or "gpt" BRAIN = "gemini" # Model names. If the bot ever says "model not found", a name got old — # check the links in README.txt for the current names and paste one here. GEMINI_MODEL = "gemini-flash-latest" # rolling alias = always current GEMINI_IMAGE_MODEL = "gemini-2.5-flash-image" # draws pictures OPENAI_MODEL = "gpt-5.6" # chat OPENAI_IMAGE_MODEL = "gpt-image-2" # draws pictures # ============================================================ # Below is the bot's machinery — you don't need to touch it. # ============================================================ def key_missing(which): """True if the key for 'gemini' or 'gpt' hasn't been pasted in yet.""" if which == "gemini": return GEMINI_API_KEY.strip() in ("", "PASTE_YOUR_GEMINI_KEY_HERE") return OPENAI_API_KEY.strip() in ("", "PASTE_YOUR_OPENAI_KEY_HERE") def need_key(which): """Print a friendly reminder if the key is missing. Returns True if missing.""" if not key_missing(which): return False if which == "gemini": print("\n⚠️ No Gemini key yet!") print(" Open mini_ai_bot.py and paste your key into GEMINI_API_KEY") print(" near the top of the file, then run me again.") print(" Free key: https://aistudio.google.com/apikey") else: print("\n⚠️ No OpenAI key yet!") print(" Open mini_ai_bot.py and paste your key into OPENAI_API_KEY") print(" near the top of the file, then run me again.") print(" (OpenAI needs a paid account: https://platform.openai.com/api-keys)") return True def friendly_http_error(who, status, message): """Turn an API error into plain words.""" message = (message or "").lower() if status in (400, 401, 403) and ("key" in message or "api key" in message or "unauthorized" in message or "invalid" in message): return ("🔑 That key didn't work. Double-check you pasted the whole key\n" " into the file with no extra spaces.") if status == 404 or "not found" in message: return ("🔍 The AI said 'model not found'. The model name in this file\n" " may be old — see README.txt for where to find the new name.") if status == 429: return "⏳ Too many requests — wait a minute and try again." if status == 403: return ("🚫 Access denied (403). For OpenAI image keys, new accounts\n" " sometimes need a one-time verification on OpenAI's site first.") return f"😕 The {who} API returned an error ({status}). Details: {message[:200]}" def gemini_error(response): try: msg = response.json().get("error", {}).get("message", "") except Exception: msg = response.text return friendly_http_error("Gemini", response.status_code, msg) def gpt_error(response): try: err = response.json().get("error", {}) msg = err.get("message", "") if isinstance(err, dict) else str(err) except Exception: msg = response.text return friendly_http_error("OpenAI", response.status_code, msg) # ---------------- brains ---------------- def ask_gemini(prompt): """Send text to Gemini, return (answer, error).""" url = ("https://generativelanguage.googleapis.com/v1beta/models/" f"{GEMINI_MODEL}:generateContent") try: r = requests.post(url, params={"key": GEMINI_API_KEY}, json={"contents": [{"parts": [{"text": prompt}]}]}, timeout=90) except Exception as e: return None, f"📡 Could not reach Google. Check your internet. ({e})" if r.status_code != 200: return None, gemini_error(r) try: parts = r.json()["candidates"][0]["content"]["parts"] text = "".join(p.get("text", "") for p in parts).strip() return (text or "(Gemini sent an empty answer — try again.)"), None except (KeyError, IndexError, TypeError): return None, "😕 Gemini answered in a way I didn't understand. Try again." def ask_gpt(prompt): """Send text to GPT, return (answer, error).""" url = "https://api.openai.com/v1/chat/completions" headers = {"Authorization": f"Bearer {OPENAI_API_KEY}"} body = {"model": OPENAI_MODEL, "messages": [{"role": "user", "content": prompt}]} try: r = requests.post(url, headers=headers, json=body, timeout=90) except Exception as e: return None, f"📡 Could not reach OpenAI. Check your internet. ({e})" if r.status_code != 200: return None, gpt_error(r) try: text = r.json()["choices"][0]["message"]["content"].strip() return (text or "(GPT sent an empty answer — try again.)"), None except (KeyError, IndexError, TypeError): return None, "😕 GPT answered in a way I didn't understand. Try again." def ask_brain(prompt): """Ask whichever brain is active. Returns (answer, error).""" if BRAIN == "gemini": return ask_gemini(prompt) return ask_gpt(prompt) # ---------------- commands ---------------- def strip_code_fences(text): """Remove ```html ... ``` wrapping if the AI added it anyway.""" t = text.strip() if t.startswith("```"): lines = t.split("\n") lines = lines[1:] # drop opening fence if lines and lines[-1].strip().startswith("```"): lines = lines[:-1] # drop closing fence t = "\n".join(lines) return t def cmd_website(idea): if need_key(BRAIN): return print(f"🌐 Asking {BRAIN} to build your website...") prompt = ( f"Write ONE complete, single-file HTML web page for this idea: {idea}\n" "Rules: output ONLY raw HTML code — no explanations, no markdown fences. " "Make it look nice with inline CSS. It must work when opened in a browser." ) html, err = ask_brain(prompt) if err: print(err) return html = strip_code_fences(html) with open("website.html", "w", encoding="utf-8") as f: f.write(html) print("✅ Saved as website.html — open it in your browser to see it!") def cmd_picture(description): if need_key(BRAIN): return print(f"🎨 Asking {BRAIN} to draw: {description} ... (this can take ~30 seconds)") try: if BRAIN == "gemini": url = ("https://generativelanguage.googleapis.com/v1beta/models/" f"{GEMINI_IMAGE_MODEL}:generateContent") body = {"contents": [{"parts": [{"text": "Draw this: " + description}]}], "generationConfig": {"responseModalities": ["TEXT", "IMAGE"]}} r = requests.post(url, params={"key": GEMINI_API_KEY}, json=body, timeout=120) if r.status_code != 200: print(gemini_error(r)) return parts = r.json()["candidates"][0]["content"]["parts"] for p in parts: inline = p.get("inlineData") or p.get("inline_data") if inline: img = base64.b64decode(inline["data"]) ext = "jpg" if "jpeg" in inline.get("mimeType", "") else "png" fname = f"picture.{ext}" with open(fname, "wb") as f: f.write(img) print(f"✅ Saved as {fname} — open it to see your picture!") return print("😕 Gemini didn't send a picture back. Try a different description.") else: url = "https://api.openai.com/v1/images/generations" headers = {"Authorization": f"Bearer {OPENAI_API_KEY}"} body = {"model": OPENAI_IMAGE_MODEL, "prompt": description, "size": "1024x1024"} r = requests.post(url, headers=headers, json=body, timeout=120) if r.status_code != 200: print(gpt_error(r)) return item = r.json()["data"][0] if item.get("b64_json"): img = base64.b64decode(item["b64_json"]) elif item.get("url"): img = requests.get(item["url"], timeout=120).content else: print("😕 OpenAI didn't send a picture back. Try again.") return with open("picture.png", "wb") as f: f.write(img) print("✅ Saved as picture.png — open it to see your picture!") except Exception as e: print(f"😕 Something went wrong while drawing. ({e})") def cmd_video(description): if need_key(BRAIN): return print(f"🎬 Asking {BRAIN} to plan your video...") prompt = ( f"Write a short, fun video plan for this idea: {description}\n" "Include: a catchy title, the total length in seconds, 3 to 5 scenes " "each with timing and on-screen text, plus a short voiceover script." ) plan, err = ask_brain(prompt) if err: print(err) return fname = "video-plan.txt" header = (f"VIDEO PLAN — made {datetime.date.today()}\n" f"Idea: {description}\n{'-' * 40}\n\n") with open(fname, "w", encoding="utf-8") as f: f.write(header + plan) print(f"✅ Video plan saved as {fname}") print(" Note: I write the plan — rendering a real video needs a") print(" separate paid video service.") def cmd_help(): print(""" Commands: ask chat with the AI website build a web page → saves website.html picture draw a picture → saves picture.png video write a video plan → saves video-plan.txt use gemini switch brain to Gemini (free key) use gpt switch brain to GPT (paid key) help show this list again quit say goodbye Tip: just typing a question works too — it counts as 'ask'. """) # ---------------- main loop ---------------- def main(): global BRAIN print("=" * 52) print(" 🤖 MINI AI BOT — type 'help' for commands, 'quit' to exit") print(f" Brain: {BRAIN} (type 'use gpt' or 'use gemini' to switch)") print("=" * 52) if key_missing("gemini") and key_missing("gpt"): print("\n👋 Hi! Before we start: paste your API key into this file") print(" (GEMINI_API_KEY near the top — free from Google AI Studio).") print(" See README.txt for baby steps.\n") while True: try: line = input("\n> ").strip() except (EOFError, KeyboardInterrupt): print("\n👋 Bye!") break if not line: continue low = line.lower() if low in ("quit", "exit", "bye"): print("👋 Bye! Have fun!") break elif low == "help": cmd_help() elif low == "use gemini": BRAIN = "gemini" print("✅ Brain switched to ✨ Gemini.") elif low == "use gpt": BRAIN = "gpt" print("✅ Brain switched to 🧠 GPT.") elif low.startswith("website "): cmd_website(line[8:].strip() or "a cool personal homepage") elif low.startswith("picture ") or low.startswith("draw "): cmd_picture(line.split(" ", 1)[1].strip()) elif low.startswith("video "): cmd_video(line[6:].strip() or "a fun short ad") elif low.startswith("ask "): if need_key(BRAIN): continue answer, err = ask_brain(line[4:].strip()) print(("\n" + (err or answer)) if (err or answer) else "") else: # plain typing counts as chatting if need_key(BRAIN): continue print("💭 thinking...") answer, err = ask_brain(line) print("\n" + (err or answer)) if __name__ == "__main__": main()
